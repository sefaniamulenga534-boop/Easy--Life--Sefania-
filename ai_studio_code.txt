import 'package:flutter/material.dart';
import 'theme.dart';

class ElLogo extends StatelessWidget {
  final double size;
  const ElLogo({super.key, this.size = 48});

  @override
  Widget build(BuildContext context) {
    return CustomPaint(
      size: Size(size, size),
      painter: _ElLogoPainter(),
    );
  }
}

class _ElLogoPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final strokeWidth = size.width * 0.14;
    final paintGreen = Paint()
      ..color = AppColors.green
      ..strokeWidth = strokeWidth
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round
      ..style = PaintingStyle.stroke;

    final paintYellow = Paint()
      ..color = AppColors.yellow
      ..strokeWidth = strokeWidth
      ..strokeCap = StrokeCap.round
      ..style = PaintingStyle.stroke;

    // Upward pointing arrow & EL dynamic fusion
    Path pathE = Path()
      ..moveTo(size.width * 0.25, size.height * 0.8)
      ..lineTo(size.width * 0.25, size.height * 0.25)
      ..lineTo(size.width * 0.55, size.height * 0.25);

    Path pathL = Path()
      ..moveTo(size.width * 0.50, size.height * 0.40)
      ..lineTo(size.width * 0.50, size.height * 0.80)
      ..lineTo(size.width * 0.85, size.height * 0.80);

    // Arrow tip pointing up
    Path arrowHead = Path()
      ..moveTo(size.width * 0.65, size.height * 0.35)
      ..lineTo(size.width * 0.85, size.height * 0.15)
      ..lineTo(size.width * 0.85, size.height * 0.40);

    canvas.drawPath(pathE, paintGreen);
    canvas.drawPath(pathL, paintYellow);
    canvas.drawPath(arrowHead, paintGreen);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}
