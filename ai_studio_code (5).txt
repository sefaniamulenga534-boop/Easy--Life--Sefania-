class Sale {
  final String id;
  final String item;
  final double amount;
  final String paymentMethod;
  final DateTime date;

  Sale({required this.id, required this.item, required this.amount, required this.paymentMethod, required this.date});
}

class SaleAdapter extends TypeAdapter<Sale> {
  @override
  final int typeId = 20;

  @override
  Sale read(BinaryReader reader) {
    return Sale(
      id: reader.readString(),
      item: reader.readString(),
      amount: reader.readDouble(),
      paymentMethod: reader.readString(),
      date: DateTime.parse(reader.readString()),
    );
  }

  @override
  void write(BinaryWriter writer, Sale obj) {
    writer.writeString(obj.id);
    writer.writeString(obj.item);
    writer.writeDouble(obj.amount);
    writer.writeString(obj.paymentMethod);
    writer.writeString(obj.date.toIso8601String());
  }
}

class Expense {
  final String id;
  final String category;
  final double amount;
  final String wallet;
  final String? receiptPath;
  final DateTime date;

  Expense({required this.id, required this.category, required this.amount, required this.wallet, this.receiptPath, required this.date});
}

class ExpenseAdapter extends TypeAdapter<Expense> {
  @override
  final int typeId = 21;

  @override
  Expense read(BinaryReader reader) {
    return Expense(
      id: reader.readString(),
      category: reader.readString(),
      amount: reader.readDouble(),
      wallet: reader.readString(),
      receiptPath: reader.read() as String?,
      date: DateTime.parse(reader.readString()),
    );
  }

  @override
  void write(BinaryWriter writer, Expense obj) {
    writer.writeString(obj.id);
    writer.writeString(obj.category);
    writer.writeDouble(obj.amount);
    writer.writeString(obj.wallet);
    writer.write(obj.receiptPath);
    writer.writeString(obj.date.toIso8601String());
  }
}

class SavingGoal {
  final String id;
  final String name;
  final double targetAmount;
  final double currentAmount;
  final DateTime deadline;

  SavingGoal({required this.id, required this.name, required this.targetAmount, required this.currentAmount, required this.deadline});

class SavingGoalAdapter extends TypeAdapter<SavingGoal> {

class ElLogo extends StatelessWidget {
  final double size;
  const ElLogo({super.key, this.size = 48});

  @override
  Widget build(BuildContext context) {
    return CustomPaint(size: Size(size, size), painter: _ElPainter());
  }
}

class _ElPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final stroke = size.width * 0.14;
    final paintG = Paint()..color = AppColors.green..strokeWidth = stroke..strokeCap = StrokeCap.round..style = PaintingStyle.stroke;
    final paintY = Paint()..color = AppColors.yellow..strokeWidth = stroke..strokeCap = StrokeCap.round..style = PaintingStyle.stroke;

    Path pathE = Path()
      ..moveTo(size.width * 0.25, size.height * 0.8)
      ..lineTo(size.width * 0.25, size.height * 0.25)
      ..lineTo(size.width * 0.55, size.height * 0.25);

    Path pathL = Path()
      ..moveTo(size.width * 0.50, size.height * 0.40)
      ..lineTo(size.width * 0.50, size.height * 0.80)
      ..lineTo(size.width * 0.85, size.height * 0.80);

    Path arrow = Path()
      ..moveTo(size.width * 0.65, size.height * 0.35)
      ..lineTo(size.width * 0.85, size.height * 0.15)
      ..lineTo(size.width * 0.85, size.height * 0.40);

    canvas.drawPath(pathE, paintG);
    canvas.drawPath(pathL, paintY);
    canvas.drawPath(arrow, paintG);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

// 6. PIN LOCK SCREEN (PIN: 1234)
// ==========================================
class PinLockScreen extends StatefulWidget {
  const PinLockScreen({super.key});

  @override
  State<PinLockScreen> createState() => _PinLockScreenState();
}

class _PinLockScreenState extends State<PinLockScreen> {
  String _enteredPin = '';

  void _key(String d) {
    if (_enteredPin.length < 4) {
      setState(() => _enteredPin += d);
      if (_enteredPin.length == 4) {
        if (_enteredPin == '1234') {
          Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => const HomeScreen()));
        } else {
          ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Incorrect PIN. Default is 1234.')));
          setState(() => _enteredPin = '');
        }
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 28),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const ElLogo(size: 50),
              const SizedBox(height: 12),
              const Text('EASY LIFE', style: TextStyle(fontSize: 22, fontWeight: FontWeight.w900, color: AppColors.green)),
              const SizedBox(height: 4),
              const Text('Enter 4-Digit Security PIN', style: TextStyle(color: AppColors.textMuted, fontSize: 13)),
              const SizedBox(height: 24),
              Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: List.generate(4, (i) => Container(
                  margin: const EdgeInsets.symmetric(horizontal: 10),
                  width: 16, height: 16,
                  decoration: BoxDecoration(shape: BoxShape.circle, color: i < _enteredPin.length ? AppColors.green : AppColors.surfaceLight),
                )),
              ),
              const SizedBox(height: 40),
              Table(
                children: [
                  TableRow(children: [_btn('1'), _btn('2'), _btn('3')]),
                  TableRow(children: [_btn('4'), _btn('5'), _btn('6')]),
                  TableRow(children: [_btn('7'), _btn('8'), _btn('9')]),
                  TableRow(children: [
                    const SizedBox(),
                    _btn('0'),
                    IconButton(icon: const Icon(Icons.backspace_outlined, color: AppColors.yellow), onPressed: () {
                      if (_enteredPin.isNotEmpty) setState(() => _enteredPin = _enteredPin.substring(0, _enteredPin.length - 1));
                    }),
                  ]),
                ],
              ),
            ],
          ),
        ),
      ),
    );
  }

  Widget _btn(String t) {
    return InkWell(
      onTap: () => _key(t),
      customBorder: const CircleBorder(),
      child: Container(height: 60, alignment: Alignment.center, child: Text(t, style: const TextStyle(fontSize: 22, fontWeight: FontWeight.bold))),
    );
  }
}

// 8. SCREEN 2: ADD SALE (VOICE INPUT & METHODS)
// ==========================================
class AddSaleScreen extends StatefulWidget {
  const AddSaleScreen({super.key});

  @override
  State<AddSaleScreen> createState() => _AddSaleScreenState();
}

class _AddSaleScreenState extends State<AddSaleScreen> {
  final _itemCtrl = TextEditingController();
  final _amtCtrl = TextEditingController();
  String _method = 'Cash';
  final SpeechToText _speech = SpeechToText();
  bool _isListening = false;

  void _listen() async {
    if (!_isListening) {
      if (await _speech.initialize()) {
        setState(() => _isListening = true);
        _speech.listen(onResult: (v) => setState(() => _itemCtrl.text = v.recognizedWords));
      }
    } else {
      setState(() => _isListening = false);
      _speech.stop();
    }
  }

  void _save() {
    final amt = double.tryParse(_amtCtrl.text) ?? 0.0;
    if (_itemCtrl.text.isNotEmpty && amt > 0) {
      final sale = Sale(id: DateTime.now().millisecondsSinceEpoch.toString(), item: _itemCtrl.text.trim(), amount: amt, paymentMethod: _method, date: DateTime.now());
      Hive.box<Sale>('sales_box').add(sale);
      Navigator.pop(context);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Add Sale')),
      body: Padding(
        padding: const EdgeInsets.all(20),
        child: Column(
          children: [
            TextField(
              controller: _itemCtrl,
              decoration: InputDecoration(
                labelText: 'Item Sold',
                suffixIcon: IconButton(icon: Icon(_isListening ? Icons.mic : Icons.mic_none, color: _isListening ? AppColors.red : AppColors.green), onPressed: _listen),
              ),
            ),
            const SizedBox(height: 14),
            TextField(controller: _amtCtrl, keyboardType: TextInputType.number, decoration: const InputDecoration(labelText: 'Amount (ZMW)', prefixText: 'K ')),
            const SizedBox(height: 14),
            DropdownButtonFormField<String>(
              value: _method,
              dropdownColor: AppColors.surface,
              decoration: const InputDecoration(labelText: 'Payment Method'),
              items: ['Cash', 'MTN', 'Airtel', 'Zamtel'].map((m) => DropdownMenuItem(value: m, child: Text(m))).toList(),
              onChanged: (v) => setState(() => _method = v!),
            ),
            const SizedBox(height: 24),
            ElevatedButton(
              style: ElevatedButton.styleFrom(backgroundColor: AppColors.green, minimumSize: const Size.fromHeight(50)),
              onPressed: _save,
              child: const Text('SAVE SALE', style: TextStyle(color: Colors.black, fontWeight: FontWeight.bold)),
            ),
            const Spacer(),
            const KitweFooter(),
          ],
        ),
      ),
    );
  }
}

// 9. SCREEN 3: ADD EXPENSE (PHOTO RECEIPT)
// ==========================================
class AddExpenseScreen extends StatefulWidget {
  const AddExpenseScreen({super.key});

  @override
  State<AddExpenseScreen> createState() => _AddExpenseScreenState();
}

class _AddExpenseScreenState extends State<AddExpenseScreen> {
  final _amtCtrl = TextEditingController();
  String _category = 'Stock';
  String _wallet = 'Cash';
  File? _receipt;

  void _pick(ImageSource src) async {
    final img = await ImagePicker().pickImage(source: src, imageQuality: 60);
    if (img != null) setState(() => _receipt = File(img.path));
  }

  void _save() {
    final amt = double.tryParse(_amtCtrl.text) ?? 0.0;
